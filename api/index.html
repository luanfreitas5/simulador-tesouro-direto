    def calculate_compound_factor(
    annual_rate: float,
    business_days: int,
    business_days_per_year: int = DEFAULT_BUSINESS_DAYS_PER_YEAR,
) -> float:
    """Capitaliza uma taxa anual em dias úteis: ``(1 + taxa) ** (du / 252)``.

    Parameters
    ----------
    annual_rate : float
        Taxa anual em fração (0.10 = 10% a.a.); deve ser maior que -1.
    business_days : int
        Dias úteis do período (não negativo).
    business_days_per_year : int, optional
        Convenção de dias úteis por ano, by default 252.

    Returns
    -------
    float
        Fator de crescimento do período.

    Raises
    ------
    InvalidInputError
        Se a taxa for ``<= -1``, os dias forem negativos ou a convenção não for positiva.

    Examples
    --------
    >>> calculate_compound_factor(0.10, 252)
    1.1
    """
    if annual_rate <= -1:
        raise InvalidInputError(f"A taxa anual deve ser maior que -100%; recebido: {annual_rate}")
    if business_days < 0:
        raise InvalidInputError("Os dias úteis não podem ser negativos.")
    if business_days_per_year <= 0:
        raise InvalidInputError("Os dias úteis por ano devem ser positivos.")
    return (1.0 + annual_rate) ** (business_days / business_days_per_year)

    def calculate_inflation_factor(
    monthly_ipca: Iterable[tuple[date, float]],
    start: date,
    end: date,
    projected_annual_rate: float,
    calendar_days_per_year: int = DEFAULT_CALENDAR_DAYS_PER_YEAR,
) -> float:
    """Acumula o IPCA em ``[start, end)``, com rateio proporcional e projeção.

    Cada mês observado entra com ``(1 + ipca) ** (dias_sobrepostos / dias_do_mês)``.
    O trecho posterior ao último mês observado usa ``projected_annual_rate``
    capitalizada por dias corridos.

    Parameters
    ----------
    monthly_ipca : Iterable[tuple[date, float]]
        Pares ``(data_do_mês, variação_em_%)``; o dia é ignorado (mês de referência).
    start : date
        Início do período (inclusivo).
    end : date
        Fim do período (exclusivo).
    projected_annual_rate : float
        IPCA anual assumido (fração) para o período sem dado observado.
    calendar_days_per_year : int, optional
        Dias corridos por ano na projeção, by default 365.

    Returns
    -------
    float
        Fator acumulado de inflação (1.0 = sem inflação).

    Raises
    ------
    InvalidInputError
        Se ``end < start`` ou a projeção for ``<= -100%``.

    Examples
    --------
    >>> ipca = [(date(2024, 1, 1), 1.0)]
    >>> round(calculate_inflation_factor(ipca, date(2024, 1, 1), date(2024, 2, 1), 0.04), 4)
    1.01
    """
    if end < start:
        raise InvalidInputError(f"A data final ({end}) é anterior à inicial ({start}).")
    if projected_annual_rate <= -1:
        raise InvalidInputError("A inflação projetada deve ser maior que -100%.")

    factor = 1.0
    observed_until = start
    for reference, pct in sorted(monthly_ipca):
        month_start = _month_start(reference)
        month_end = _next_month_start(reference)
        overlap_days = (min(end, month_end) - max(start, month_start)).days
        if overlap_days > 0:
            month_days = (month_end - month_start).days
            factor *= (1.0 + pct / 100.0) ** (overlap_days / month_days)
        observed_until = max(observed_until, month_end)

    projected_days = max((end - observed_until).days, 0)
    return factor * (1.0 + projected_annual_rate) ** (projected_days / calendar_days_per_year)

    def calculate_selic_factor(
    daily_selic: Sequence[tuple[date, float]],
    start: date,
    end: date,
    projected_annual_rate: float,
    holidays: Iterable[date] = (),
    business_days_per_year: int = DEFAULT_BUSINESS_DAYS_PER_YEAR,
) -> float:
    """Acumula a Selic diária em ``[start, end)`` e projeta o trecho restante.

    Parameters
    ----------
    daily_selic : Sequence[tuple[date, float]]
        Pares ``(data, taxa_diária_em_%)`` (série SGS 11, dias úteis).
    start : date
        Início do período (inclusivo).
    end : date
        Fim do período (exclusivo).
    projected_annual_rate : float
        Selic anual assumida (fração) após o último dado observado.
    holidays : Iterable[date], optional
        Feriados para a contagem de dias úteis da projeção, by default vazio.
    business_days_per_year : int, optional
        Convenção de dias úteis por ano, by default 252.

    Returns
    -------
    float
        Fator acumulado da Selic no período.

    Raises
    ------
    InvalidInputError
        Se ``end < start`` ou a projeção for inválida.

    Examples
    --------
    >>> selic = [(date(2024, 1, 8), 0.04)]
    >>> round(calculate_selic_factor(selic, date(2024, 1, 8), date(2024, 1, 9), 0.10), 6)
    1.0004
    """
    if end < start:
        raise InvalidInputError(f"A data final ({end}) é anterior à inicial ({start}).")

    factor = 1.0
    last_observed = start - timedelta(days=1)
    for reference, pct in daily_selic:
        if start <= reference < end:
            factor *= 1.0 + pct / 100.0
        last_observed = max(last_observed, min(reference, end - timedelta(days=1)))

    projection_start = last_observed + timedelta(days=1)
    if projection_start >= end:
        return factor
    projected_days = count_business_days(projection_start, end, holidays)
    return factor * calculate_compound_factor(
        projected_annual_rate, projected_days, business_days_per_year
    )

    def ensure_snapshot(
    source_dir: Path,
    filename: str,
    fetch: Callable[[], bytes],
    url: str,
    ttl_hours: float = 24.0,
    force: bool = False,
    now: datetime | None = None,

    """Garante um snapshot local do arquivo e retorna seu caminho (idempotente).

    - Dentro do TTL e sem ``force``: reutiliza o snapshot, sem rede.

    source_dir : Path

    filename : str
        Nome do arquivo dentro do snapshot.
    fetch : Callable[[], bytes]
        Função sem argumentos que baixa o conteúdo.
    url : str
        URL de origem (registrada no manifest).
    ttl_hours : float, optional
        Validade do cache em horas, by default 24.
    force : bool, optional
        Ignora o cache e baixa novamente, by default False.
    now : datetime | None, optional
        Instante atual (injetável para testes); por padrão, UTC agora.

    Returns
    -------
    Path
        Caminho do arquivo no snapshot vigente.

    ------
    DataIngestionError
        Se o download falhar e não houver snapshot anterior.

    Examples
    --------
    >>> ensure_snapshot(Path("data/bronze/x"), "a.csv", lambda: b"a", "http://x")  # doctest: +SKIP
    """
    now = now or datetime.now(timezone.utc)
    latest = find_latest_snapshot(source_dir)

    if latest is not None and not force and _is_fresh(source_dir, ttl_hours, now):
        logger.info("Cache válido para %s; nenhum download necessário.", source_dir.name)
        return latest / filename

    try:
        content = fetch()
    except DataIngestionError:
        if latest is None:
            raise
        logger.warning("Download falhou; usando o snapshot local de %s.", source_dir.name)
        return latest / filename

    digest = hash_bytes(content)
    source_dir.mkdir(parents=True, exist_ok=True)
    (source_dir / LAST_CHECK_FILE).write_text(now.isoformat(), encoding="utf-8")

    if latest is not None and _read_json(latest / MANIFEST_FILE).get("sha256") == digest:
        logger.info("Conteúdo de %s inalterado; snapshot mantido.", source_dir.name)
        return latest / filename

    path = _write_snapshot(source_dir, filename, content, url, digest, now)
    logger.info("Novo snapshot gravado: %s", path)
    return path

    def split_date_windows(start: date, end: date, max_years: int) -> list[tuple[date, date]]:
    """Divide ``[start, end]`` em janelas de anos-calendário de até ``max_years`` anos.

    A API do SGS limita consultas de séries diárias a janelas de 10 anos.

    Parameters
    ----------
    start : date
        Data inicial.
    end : date
        Data final.
    max_years : int
        Máximo de anos-calendário por janela (>= 1).

    Returns
    -------
    list[tuple[date, date]]
        Janelas contíguas e sem sobreposição (vazia se ``end < start``).

    Examples
    --------
    >>> windows = split_date_windows(date(2002, 1, 1), date(2012, 6, 30), 9)
    >>> [(a.isoformat(), b.isoformat()) for a, b in windows]
    [('2002-01-01', '2010-12-31'), ('2011-01-01', '2012-06-30')]
    """
    windows: list[tuple[date, date]] = []
    current = start
    while current <= end:
        window_end = min(date(current.year + max_years - 1, 12, 31), end)
        windows.append((current, window_end))
        current = window_end + timedelta(days=1)
    return windows

    """Constrói a curva de juros: taxa de compra por vencimento em cada data.

    Mantém apenas os títulos solicitados com taxa de compra disponível e
    vencimento posterior à data de referência, e adiciona ``years_to_maturity``
    (dias corridos / 365,25) para o eixo X da curva.

    Parameters
    ----------
    silver : pl.DataFrame

    -------

        Colunas ``reference_date``, ``bond_type``, ``maturity_date``,
        ``years_to_maturity``, ``buy_rate`` e ``sell_rate`` (taxas em % a.a.),

    Examples
    --------
    >>> curve = build_yield_curve(silver)  # doctest: +SKIP
    """
    return (
        silver.lazy()
        .filter(pl.col("bond_type").is_in(list(bond_types)) & pl.col("buy_rate").is_not_null())
        .with_columns(
            (
                (pl.col("maturity_date") - pl.col("reference_date")).dt.total_days()
                / DAYS_PER_YEAR_CURVE
            ).alias("years_to_maturity")
        )
        .filter(pl.col("years_to_maturity") > 0)
        .select(list(GOLD_CURVE_COLUMNS))
        .sort(["bond_type", "reference_date", "maturity_date"])
        .collect()
    )
